  { name: 'dot-segment', argv: (r) => ['allow-execution', path.join(r, '..', '.') + '/' + path.basename(r) + '/.', '--revision', REV, '--declare', D, '--json'] },

it('differential', async () => {
  const out: unknown[] = [];
  for (const [pname, plant] of Object.entries(plants)) {
    for (const v of variants) {
      const run = runDirectory(v.owner ?? true, v.dirName ?? RUN_ID);
      const root = path.dirname(run);
      v.pre?.(run);
      if (v.name === 'recorded-already') {
        const first = await cli(['allow-execution', run, '--revision', REV, '--declare', D, '--json']);
        void first;
      }
      if (plant) plant(path.join(run, 'launched.json'), run);
      const before = fs.existsSync(path.join(run, 'execution-choice.json')) ? fs.readFileSync(path.join(run, 'execution-choice.json'), 'utf8') : null;
      const r = await cli(v.argv(run), v.over ?? {});
      const after = fs.existsSync(path.join(run, 'execution-choice.json')) ? fs.readFileSync(path.join(run, 'execution-choice.json'), 'utf8') : null;
      const norm = (s: string) => s.split(root).join('<ROOT>');
      out.push({ plant: pname, variant: v.name, code: r.code, stdout: norm(r.stdout), stderr: norm(r.stderr), wrote: before !== after, choice: after === null ? null : norm(after) });
    }
  }
  // Lexical versus physical `..` through a symlinked component.
  for (const [where, launchedAt] of [['physical', 'physical'], ['lexical', 'lexical'], ['neither', 'none'], ['both', 'both']] as const) {
    const base = tempDir('dossier-lex-');
    // physical: <base>/p/inner is where link points; <base>/p/run-a is the physical target of link/../run-a.
    // lexical: <base>/run-a is what path.resolve(<base>/link/../run-a) names.
    fs.mkdirSync(path.join(base, 'p', 'inner'), { recursive: true });
    const mk = (dir: string) => {
      fs.mkdirSync(dir);
      const src = runDirectory();
      fs.copyFileSync(path.join(src, 'run.json'), path.join(dir, 'run.json'));
    };
    const phys = path.join(base, 'p', RUN_ID), lex = path.join(base, RUN_ID);
    mk(phys); mk(lex);
    fs.symlinkSync(path.join(base, 'p', 'inner'), path.join(base, 'link'));
    if (launchedAt === 'physical' || launchedAt === 'both') writeLaunchedMarker(phys, RUN_ID, NOW);
    if (launchedAt === 'lexical' || launchedAt === 'both') writeLaunchedMarker(lex, RUN_ID, NOW);
    const arg = `${base}/link/../${RUN_ID}`;
    const r = await cli(['allow-execution', arg, '--revision', REV, '--declare', D, '--json']);
    const norm = (s: string) => s.split(base).join('<BASE>');
    out.push({ plant: `symlink-dotdot:${where}`, variant: 'all-good', code: r.code, stdout: norm(r.stdout), stderr: norm(r.stderr),
      wrotePhys: fs.existsSync(path.join(phys, 'execution-choice.json')), wroteLex: fs.existsSync(path.join(lex, 'execution-choice.json')) });
  }
  wf(process.env.DIFF_OUT!, JSON.stringify(out, null, 1));
  expect(out.length).toBeGreaterThan(0);
});
